function Get-Url([string]$u) {
  try {
    $r = Invoke-WebRequest -Uri $u -UseBasicParsing -TimeoutSec 8
    \"GET $u status=$($r.StatusCode) len=$($r.RawContentLength)\"
  } catch {
    \"GET $u FAIL $($_.Exception.Message)\"
  }
}
